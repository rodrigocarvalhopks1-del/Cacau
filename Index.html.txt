    <script>
        // Lista de mensagens românticas em português
        const romanticMessages = [
            "Eu amo o jeito que o seu sorriso ilumina qualquer ambiente! ☀️",
            "Você é a melhor parte de todos os meus dias. Te amo infinito! 💖",
            "Abraço virtual apertado enviado com sucesso! Sinta meu carinho agora. 🤗",
            "Amo cada detalhe seu, do seu jeitinho até a sua voz. 🥰",
            "Se eu pudesse escolher de novo, escolheria você mil vezes! 💍",
            "Programei cada linha dessa página pensando em como sou sortudo por ter você. 💻❤️",
            "Você é meu porto seguro, minha melhor amiga e o amor da minha vida. 🌹",
            "Só passando para lembrar: você é linda, incrível e eu sou louco por você! 😍",
            "Meu lugar favorito no mundo inteiro é dentro do seu abraço. 🏡💕",
            "Obrigado por fazer minha vida tão mais feliz e colorida! ✨"
        ];

        let lastMessageIndex = -1;
        let hugCount = 0;

        // Inicia os corações flutuantes no plano de fundo ao carregar a página
        window.addEventListener('DOMContentLoaded', () => {
            createBackgroundHearts();
        });

        // Cria corações suaves subindo no fundo da tela
        function createBackgroundHearts() {
            const container = document.getElementById('floating-hearts-container');
            const heartSymbols = ['❤️', '💖', '💕', '🌸', '💗', '✨'];
            const totalHearts = 16;

            for (let i = 0; i < totalHearts; i++) {
                const heart = document.createElement('div');
                heart.classList.add('bg-heart');
                heart.innerText = heartSymbols[Math.floor(Math.random() * heartSymbols.length)];
                
                // Posição horizontal aleatória
                heart.style.left = Math.random() * 100 + 'vw';
                // Tamanho variado
                const size = Math.random() * 14 + 14;
                heart.style.fontSize = size + 'px';
                // Duração e atraso aleatórios para movimento orgânico
                const duration = Math.random() * 8 + 8;
                heart.style.animationDuration = duration + 's';
                heart.style.animationDelay = (Math.random() * 10) + 's';

                container.appendChild(heart);
            }
        }

        // Função para abrir a carta de amor
        function openLetter(event) {
            const closedCard = document.getElementById('envelope-closed');
            const openedCard = document.getElementById('envelope-opened');

            // Dispara explosão de corações no toque
            spawnHeartBurst(event.clientX || window.innerWidth / 2, event.clientY || window.innerHeight / 3, 18);
            playSoftChime();

            closedCard.classList.add('hidden');
            openedCard.classList.remove('hidden');
        }

        // Função para fechar a carta de amor
        function closeLetter() {
            const closedCard = document.getElementById('envelope-closed');
            const openedCard = document.getElementById('envelope-opened');

            openedCard.classList.add('hidden');
            closedCard.classList.remove('hidden');
        }

        // Função acionada ao clicar no botão de amor/abraço
        function triggerLove(event) {
            const messageEl = document.getElementById('love-message');
            const counterEl = document.getElementById('hug-counter');

            // Escolhe uma mensagem diferente da anterior
            let randomIndex;
            do {
                randomIndex = Math.floor(Math.random() * romanticMessages.length);
            } while (randomIndex === lastMessageIndex && romanticMessages.length > 1);
            
            lastMessageIndex = randomIndex;

            // Animação suave de troca de texto
            messageEl.style.opacity = '0';
            messageEl.style.transform = 'scale(0.95)';

            setTimeout(() => {
                messageEl.innerText = romanticMessages[randomIndex];
                messageEl.style.opacity = '1';
                messageEl.style.transform = 'scale(1)';
            }, 150);

            // Atualiza contador
            hugCount++;
            counterEl.innerText = hugCount;

            // Coordenadas do toque ou centro do botão
            const rect = event.currentTarget.getBoundingClientRect();
            const x = event.clientX || (rect.left + rect.width / 2);
            const y = event.clientY || (rect.top + rect.height / 2);

            // Explosão de corações + som suave
            spawnHeartBurst(x, y, 14);
            playSoftChime();
        }

        // Cria explosão de mini corações a partir do ponto de clique
        function spawnHeartBurst(x, y, count) {
            const emojis = ['❤️', '💖', '💕', '🥰', '💋', '✨'];

            for (let i = 0; i < count; i++) {
                const particle = document.createElement('span');
                particle.classList.add('burst-heart');
                particle.innerText = emojis[Math.floor(Math.random() * emojis.length)];
                particle.style.left = x + 'px';
                particle.style.top = y + 'px';
                particle.style.fontSize = (Math.random() * 12 + 16) + 'px';

                // Direção aleatória da explosão
                const angle = Math.random() * Math.PI * 2;
                const distance = Math.random() * 110 + 40;
                const tx = Math.cos(angle) * distance + 'px';
                const ty = (Math.sin(angle) * distance - 40) + 'px'; // tende levemente para cima
                const rot = (Math.random() * 360 - 180) + 'deg';

                particle.style.setProperty('--tx', tx);
                particle.style.setProperty('--ty', ty);
                particle.style.setProperty('--rot', rot);

                document.body.appendChild(particle);

                // Remove o elemento após terminar a animação
                setTimeout(() => {
                    particle.remove();
                }, 1000);
            }
        }

        // Sintetizador leve via Web Audio API para um som mágico e sutil ao clicar (sem arquivos externos)
        function playSoftChime() {
            try {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                if (!AudioCtx) return;
                
                const ctx = new AudioCtx();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();

                // Notas suaves de harpa/sininho
                const notes = [523.25, 659.25, 783.99, 1046.50]; // Dó, Mi, Sol, Dó agudo
                const freq = notes[Math.floor(Math.random() * notes.length)];

                osc.type = 'sine';
                osc.frequency.setValueAtTime(freq, ctx.currentTime);
                
                gain.gain.setValueAtTime(0.08, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + 0.5);

                osc.connect(gain);
                gain.connect(ctx.destination);

                osc.start();
                osc.stop(ctx.currentTime + 0.5);
            } catch (e) {
                // Ignora silenciosamente se o navegador bloquear áudio automático
            }
        }
